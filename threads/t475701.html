Starting a new thread for fun!Feel free to take some code if wanted!(My real name removed from comments for obvious reasons)Coded for my fiancée, she had many downloads where the file names had underscores instead of spaces, so I created this little thing for her to automatically replace all underscores by spaces:'*******************************************************************************'*'* Coded by : Meru Otonashi'*'* History : 2011-10-03, Project creation.'*'* Project Name : Underscore Replacer'*'* Module Name : Underscore Replacer.vbs'*'* Description : Replaces all instances of underscores by spaces in file'* names.'*'* Usage : Simply copy the vbs file to the folder needed and run'* it.'*'* Notes : This script will ignore folders and hidden files.'*'*******************************************************************************Option ExplicitCall MainWscript.QuitPublic Sub Main 'Declares variables Dim objFile, objFSO, objShell Dim strInitial, strAfter 'Sets the initial values Const strTempFile = "rename.tmp" Set objShell = CreateObject("Wscript.Shell") Set objFSO = CreateObject("Scripting.FileSystemObject") 'Runs shell command to list file names into a temporary file. objShell.Run "cmd /c dir /a:-d-h /b > " & strTempFile, 0, True 'Verifies if the temp file was created successfully If objFSO.FileExists(strTempFile) Then Set objFile = objFSO.OpenTextFile(strTempFile, 1) Do Until objFile.AtEndOfStream strInitial = Trim(objFile.Readline) strAfter = Replace(strInitial, "_", " ") objShell.Run "cmd /c ren """ & strInitial & """ """ & strAfter & """", 0, True Loop objFile.Close objFSO.DeleteFile strTempFile, True Else MsgBox "There was an error creating the temp file. Make sure you have file creation rights to the folder.", 16, "Underscore Replacer" Wscript.Quit 1 End If MsgBox "Underscores were successfully replaced.", 64, "Underscore Replacer" 'Clearing system memory Set objFile = Nothing Set objFSO = Nothing Set objShell = NothingEnd SubAt my current job, in our computer population, many computers do not have a same profile, and we're trying to normalize all the workstations we have.This is a quick script that will connect to all workstations identified in a list file and identify the version of an executable file that you fill in an input box, and then generates a log file with the details.'*******************************************************************************'*'* Par : Meru Otonashi'*'* Date : 2011-30-09, Création du projet.'*'* Nom du Projet : Recherche de version de fichier.'*'* Nom du Module : [GetFileVersion.vbs]'*'* Description : Créé un log de l'état d'un package sur les postes.'*'* Référence : Sans objet.'*'* Note : Aucune.'*'*******************************************************************************Option ExplicitCall MainWscript.QuitPublic Sub Main() Dim objFile, objFSO, objShell Dim strPath, strComputer, strFile, strVersion Dim boolPing Set objShell = WScript.CreateObject("WScript.Shell") Set objFSO = CreateObject("Scripting.FileSystemObject") strPath = ".Computers.lst" If objFSO.FileExists(strPath) Then strFile = InputBox("Entrez le lieu exacte du fichier sur les postes." & vbNewLine & "Exemple: C$WindowsSystem32wscript.exe" , "Enquête de version de fichier", "C$") If strFile = "" Then Wscript.Quit Else WriteLog "Liste de version du fichier: " & strFile, 0 Set objFile = objFSO.OpenTextFile(strPath, 1) Do Until objFile.AtEndOfStream strComputer = Trim(objFile.Readline) boolPing = Not CBool(objShell.run("ping -n 1 " & strComputer, 0, True)) If boolPing Then If objFSO.FileExists("" & strComputer & "" & strFile) Then strVersion = objFSO.GetFileVersion("" & strComputer & "" & strFile) WriteLog "[" & strComputer & "] : " & strVersion, 1 Else WriteLog "[" & strComputer & "] : Fichier introuvable.", 1 End If Else WriteLog "[" & strComputer & "] : Poste de travail ne réponds pas.", 1 End If Loop Set objFile = Nothing End If MsgBox("Opération terminée.") Else MsgBox("Fichier 'Computers.lst' introuvable.") End If Set objFSO = Nothing Set objShell = NothingEnd SubPrivate Sub WriteLog(strText, intTimeStamp) 'Declarations Dim strLogFolder, strLogFile, strLineToWrite Dim objFSO 'Parameter sets Set objFSO = CreateObject("Scripting.FileSystemObject") strLogFolder = ".results.log" strLineToWrite = "" 'Error check On Error Resume Next 'Creates log file if not existant If Not objFSO.FileExists(strLogFolder) Then objFSO.CreateTextFile(strLogFolder) End If 'Write log to file Set strLogFile = objFSO.OpenTextFile(strLogFolder, 8, True) Select Case intTimeStamp Case "0" strLineToWrite = strText Case "1" strLineToWrite = "[" + FormatDateTime(Now(), 0) + "] " + strText End Select If Not strLineToWrite = "" Then strLogFile.WriteLine strLineToWrite End If strLogFile.Close 'Liberates ressources strLogFolder = Empty strLogFile = Empty strLineToWrite = Empty Set objFSO = NothingEnd Sub
